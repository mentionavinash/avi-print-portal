-- ========================================== -- AVIPRINT ORDERS TABLE -- ========================================== create table if not exists public.orders ( id uuid primary key default gen_random_uuid(), order_number text not null unique, user_id uuid not null references auth.users(id) on delete cascade, service_name text not null, service_details text, order_date timestamptz not null default now(), status text not null default 'Pending' check ( status in ( 'Pending', 'Processing', 'Completed', 'Cancelled' ) ), amount numeric(10,2) not null default 0 check (amount >= 0), created_at timestamptz not null default now() ); -- ========================================== -- INDEX -- ========================================== create index if not exists orders_user_id_idx on public.orders(user_id); -- ========================================== -- ENABLE ROW LEVEL SECURITY -- ========================================== alter table public.orders enable row level security; -- ========================================== -- DELETE OLD POLICIES IF THEY EXIST -- ========================================== drop policy if exists "Users can view their own orders" on public.orders; drop policy if exists "Users can create their own orders" on public.orders; -- ========================================== -- SELECT POLICY -- USER CAN ONLY SEE OWN ORDERS -- ========================================== create policy "Users can view their own orders" on public.orders for select to authenticated using ( (select auth.uid()) = user_id ); -- ========================================== -- INSERT POLICY -- USER CAN ONLY CREATE OWN ORDERS -- ========================================== create policy "Users can create their own orders" on public.orders for insert to authenticated with check ( (select auth.uid()) = user_id ); -- ========================================== -- DATA API PERMISSIONS -- ========================================== grant select, insert on public.orders to authenticated;
